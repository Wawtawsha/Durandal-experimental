```yaml
# Azure Pipeline for Fabric Deployment
# Generated by Claude Code for PASS Summit Demo

trigger:
  branches:
    include:
    - main
    - develop
  paths:
    include:
    - generated/*.json

pool:
  vmImage: 'windows-latest'

variables:
  - group: Fabric-Deployment-Variables
  - name: workspaceId
    value: '64c9ba43-a51a-4eb6-ba51-18748a91e23d'
  - name: deploymentEnvironment
    value: $[variables['Build.SourceBranchName']]

stages:
- stage: Validate
  displayName: 'Validate Pipeline JSONs'
  jobs:
  - job: ValidateJSON
    displayName: 'Validate JSON Structure'
    steps:
    - task: PowerShell@2
      displayName: 'Run JSON Validation'
      inputs:
        filePath: '$(System.DefaultWorkingDirectory)/scripts/validate-pipeline.ps1'
        arguments: '-PipelinePath $(System.DefaultWorkingDirectory)/generated'

- stage: DeployToFabric
  displayName: 'Deploy to Fabric Workspace'
  dependsOn: Validate
  condition: succeeded()
  jobs:
  - deployment: DeployPipelines
    displayName: 'Deploy Pipelines to Fabric'
    environment: 'Fabric-Production'
    strategy:
      runOnce:
        deploy:
          steps:
          - checkout: self

          - task: AzurePowerShell@5
            displayName: 'Authenticate to Fabric'
            inputs:
              azureSubscription: 'Fabric-Service-Connection'
              ScriptType: 'InlineScript'
              Inline: |
                $token = Get-AzAccessToken -ResourceUrl "https://api.fabric.microsoft.com"
                Write-Host "##vso[task.setvariable variable=FabricToken;issecret=true]$($token.Token)"

          - task: PowerShell@2
            displayName: 'Deploy Pipelines'
            inputs:
              filePath: '$(System.DefaultWorkingDirectory)/scripts/deploy-fabric-pipeline.ps1'
              arguments: |
                -WorkspaceId $(workspaceId)
                -Token $(FabricToken)
                -PipelinePath $(System.DefaultWorkingDirectory)/generated

          - task: PowerShell@2
            displayName: 'Verify Deployment'
            inputs:
              targetType: 'inline'
              script: |
                Write-Host "Deployment completed successfully"
                Write-Host "Workspace: $(workspaceId)"
                Write-Host "Environment: $(deploymentEnvironment)"
                Write-Host "Deployed at: $(Get-Date -Format 'yyyy-MM-dd HH:mm:ss')"

- stage: PostDeployment
  displayName: 'Post-Deployment Tasks'
  dependsOn: DeployToFabric
  condition: succeeded()
  jobs:
  - job: UpdateDocumentation
    displayName: 'Update Deployment Log'
    steps:
    - task: PowerShell@2
      displayName: 'Log Deployment'
      inputs:
        targetType: 'inline'
        script: |
          $deploymentLog = @{
            Timestamp = Get-Date -Format 'yyyy-MM-dd HH:mm:ss'
            Environment = "$(deploymentEnvironment)"
            WorkspaceId = "$(workspaceId)"
            PipelineCount = (Get-ChildItem $(System.DefaultWorkingDirectory)/generated/*.json).Count
            CommitId = "$(Build.SourceVersion)"
            DeployedBy = "Claude Code Automation"
          }
          $deploymentLog | ConvertTo-Json | Out-File -FilePath "$(System.DefaultWorkingDirectory)/deployment-log.json"

### TASK-010: Create Deployment PowerShell Script

**File**: `scripts/deploy-fabric-pipeline.ps1`

```powershell
<#
.SYNOPSIS
    Deploys generated pipelines to Microsoft Fabric workspace
.DESCRIPTION
    This script is called by Azure DevOps to deploy Claude Code generated pipelines
.PARAMETER WorkspaceId
    Target Fabric workspace ID
.PARAMETER Token
    Bearer token for Fabric API authentication
.PARAMETER PipelinePath
    Path to generated pipeline JSON files
#>

param(
    [Parameter(Mandatory=$true)]
    [string]$WorkspaceId,

    [Parameter(Mandatory=$true)]
    [string]$Token,

    [Parameter(Mandatory=$true)]
    [string]$PipelinePath
)

# Set up headers for API calls
$headers = @{
    "Authorization" = "Bearer $Token"
    "Content-Type" = "application/json"
}

# Base URI for Fabric API
$baseUri = "https://api.fabric.microsoft.com/v1/workspaces/$WorkspaceId"

# Function to check if pipeline exists
function Test-PipelineExists {
    param($PipelineName)

    try {
        $existingPipelines = Invoke-RestMethod `
            -Uri "$baseUri/pipelines" `
            -Method Get `
            -Headers $headers

        return $existingPipelines.value | Where-Object { $_.name -eq $PipelineName }
    }
    catch {
        return $null
    }
}

# Get all pipeline JSON files
$pipelineFiles = Get-ChildItem -Path $PipelinePath -Filter "*.json"

foreach ($file in $pipelineFiles) {
    Write-Host "Processing pipeline: $($file.Name)" -ForegroundColor Cyan

    # Read pipeline content
    $pipelineContent = Get-Content $file.FullName -Raw | ConvertFrom-Json

    # Extract pipeline name from content or filename
    $pipelineName = if ($pipelineContent.name) {
        $pipelineContent.name
    } else {
        [System.IO.Path]::GetFileNameWithoutExtension($file.Name)
    }

    # Check if pipeline exists
    $existingPipeline = Test-PipelineExists -PipelineName $pipelineName

    try {
        if ($existingPipeline) {
            # Update existing pipeline
            Write-Host "Updating existing pipeline: $pipelineName" -ForegroundColor Yellow

            $updateUri = "$baseUri/pipelines/$($existingPipeline.id)"
            $response = Invoke-RestMethod `
                -Uri $updateUri `
                -Method Patch `
                -Headers $headers `
                -Body ($pipelineContent | ConvertTo-Json -Depth 100)

            Write-Host "Successfully updated pipeline: $pipelineName" -ForegroundColor Green
        }
        else {
            # Create new pipeline
            Write-Host "Creating new pipeline: $pipelineName" -ForegroundColor Yellow

            $createUri = "$baseUri/pipelines"
            $response = Invoke-RestMethod `
                -Uri $createUri `
                -Method Post `
                -Headers $headers `
                -Body ($pipelineContent | ConvertTo-Json -Depth 100)

            Write-Host "Successfully created pipeline: $pipelineName" -ForegroundColor Green
        }

        # Log the deployment
        $logEntry = @{
            Timestamp = Get-Date -Format "yyyy-MM-dd HH:mm:ss"
            Pipeline = $pipelineName
            Action = if ($existingPipeline) { "Updated" } else { "Created" }
            WorkspaceId = $WorkspaceId
            Status = "Success"
        }

        $logEntry | ConvertTo-Json | Add-Content -Path "$PipelinePath\..\deployment-history.log"
    }
    catch {
        Write-Error "Failed to deploy pipeline: $pipelineName"
        Write-Error $_.Exception.Message

        # Log the error
        $errorEntry = @{
            Timestamp = Get-Date -Format "yyyy-MM-dd HH:mm:ss"
            Pipeline = $pipelineName
            Error = $_.Exception.Message
            WorkspaceId = $WorkspaceId
            Status = "Failed"
        }

        $errorEntry | ConvertTo-Json | Add-Content -Path "$PipelinePath\..\deployment-errors.log"

        # Continue with next pipeline
        continue
    }
}

Write-Host "`nDeployment Summary:" -ForegroundColor Magenta
Write-Host "Total pipelines processed: $($pipelineFiles.Count)" -ForegroundColor Cyan
Write-Host "Check deployment-history.log for details" -ForegroundColor Gray
```

### TASK-011: Create Pattern Merger Script

**File**: `scripts/merge-pattern-parameters.ps1`

```powershell
<#
.SYNOPSIS
    Merges pipeline patterns with parameters to generate deployable pipelines
.DESCRIPTION
    Core logic for Claude Code pattern + parameter combination
#>

param(
    [Parameter(Mandatory=$true)]
    [string]$PatternPath,

    [Parameter(Mandatory=$true)]
    [string]$ParameterPath,

    [Parameter(Mandatory=$true)]
    [string]$OutputPath
)

# Load pattern and parameters
$pattern = Get-Content $PatternPath -Raw | ConvertFrom-Json
$parameters = Get-Content $ParameterPath -Raw | ConvertFrom-Json

# Function to replace placeholders with actual values
function Merge-PatternWithParameters {
    param($Pattern, $Parameters)

    $json = $Pattern | ConvertTo-Json -Depth 100

    foreach ($param in $Parameters.PSObject.Properties) {
        $placeholder = "{{$($param.Name)}}"
        $json = $json -replace [regex]::Escape($placeholder), $param.Value
    }

    return $json | ConvertFrom-Json
}

# Perform the merge
$mergedPipeline = Merge-PatternWithParameters -Pattern $pattern -Parameters $parameters

# Add metadata
$mergedPipeline | Add-Member -NotePropertyName "generatedBy" -NotePropertyValue "Claude Code Pattern Engine"
$mergedPipeline | Add-Member -NotePropertyName "generatedAt" -NotePropertyValue (Get-Date -Format "yyyy-MM-dd HH:mm:ss")
$mergedPipeline | Add-Member -NotePropertyName "patternSource" -NotePropertyValue $PatternPath
$mergedPipeline | Add-Member -NotePropertyName "parameterSource" -NotePropertyValue $ParameterPath

# Save the merged pipeline
$mergedPipeline | ConvertTo-Json -Depth 100 | Out-File -FilePath $OutputPath

Write-Host "Pipeline generated successfully: $OutputPath" -ForegroundColor Green
```

# Claude Code Pipeline Generation Instructions

## When User Requests Pipeline Generation:

1. **Identify Pattern Type**
   - Ask: "Which pattern? (copy/incremental/full-load/custom)"
   - Load pattern from: `/patterns/base/[pattern-name].json`

2. **Gather Parameters**
   - Ask for: Source, Sink, Mappings, Environment
   - Create parameter file in: `/parameters/temp/`

3. **Generate Pipeline**
   ```bash
   # Run the merger script
   ./scripts/merge-pattern-parameters.ps1 \
     -PatternPath ./patterns/base/[pattern].json \
     -ParameterPath ./parameters/temp/[params].json \
     -OutputPath ./generated/[pipeline-name].json
   ```

4. **Commit to Azure DevOps**
   ```bash
   git add ./generated/[pipeline-name].json
   git commit -m "[Claude Code] Generated [pipeline-name] using [pattern] pattern"
   git push origin main
   ```

5. **Trigger Deployment**
   - Push automatically triggers Azure Pipeline
   - Monitor at: https://dev.azure.com/[org]/[project]/_build

## Pattern Template Variables

Use these placeholders in patterns:
- `{{sourcePath}}` - Source data location
- `{{sourceConnection}}` - Source connection ID
- `{{sinkDatabase}}` - Target database
- `{{sinkSchema}}` - Target schema
- `{{sinkTable}}` - Target table
- `{{workspaceId}}` - Fabric workspace ID
- `{{environment}}` - dev/test/prod

## Example Generation Flow

User: "Create weather pipeline for production"

Claude Code Actions:
1. Load: `/patterns/base/incremental-load-pattern.json`
2. Load: `/parameters/environments/prod.parameters.json`
3. Merge: Create `/generated/weather-pipeline-prod.json`
4. Commit: Push to Azure DevOps
5. Deploy: Automatic via Azure Pipeline
6. Confirm: "Pipeline deployed to Fabric workspace"
